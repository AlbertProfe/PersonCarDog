Service.java
public static boolean buyCar(Person buyer, Person seller, int price, Repository repo) { // checks omitted for clarity Car car = seller.getCar(); buyer.setCar(car); seller.setCar(null); CarTransaction transaction = new CarTransaction(buyer, seller, new Date(), car, "Car sold for " + price); repo.addCarTransaction(transaction); return true; }
