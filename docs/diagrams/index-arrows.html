Service.java
public static boolean buyCar(
   Person seller, int price) {

  // checks omitted for clarity

  Car car = seller.getCar();
  buyer.setCar(car);
  seller.setCar(null);

  CarTransaction transaction =
     new CarTransaction(buyer, seller,
        car, "Car sold for " + price);

  repo.addCarTransaction(transaction);
  return true;
}
